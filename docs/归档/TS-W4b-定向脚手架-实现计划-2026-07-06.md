**Goal:** 给 TS harness 装上「定向脚手架」——让 agent 会用**任务清单(todo)**跟多步进度、每轮吃到**系统提醒(system-reminder)**、跑一半能被老板**插话纠偏(steering)**不打断在做的工具、以及**计划模式**下只规划不动手时收到明确指令。

**Architecture:** 四件事,都挂在主循环上:①**todo** = `TodoWrite` 工具写 `ctx.todos`(+ 任何工具调用可带 `task_progress` markdown 清单的 Cline Focus-Chain 内联路),循环在 todos 变化时吐 `todo_update` 事件;②**system-reminder** = `<system-reminder>…</system-reminder>` XML 包壳(照 cc-haha),**只往消息尾部 append、绝不改早前历史**(prompt-cache 前缀稳定)、且**只在一批工具全配对后或收尾时注入**(否则破 tool_call_id 配对→400);③**steering** = `ctx.steerInbox` FIFO(将来 HTTP 路由 push 进去),循环在**两个安全点**(每批工具后 + 收尾边界)取空、作 `[用户补充/纠偏]` 用户消息灌进去、吐 `steering` 事件,并给轮次上限**续命**(防跑不完也防跑不停)——为此**主循环从 `for` 改 `while`(可变 turnsLimit)**;④**plan 模式说明** = 权限判定(plan 档跳过会动手的工具)W4a 已做,W4b 补「在 plan 档每轮注入一条计划模式 system-reminder」告诉模型只规划。**我们不中断在跑的工具**(cc-haha 能中断,我们只排队到下个安全点——保留非中断语义,红线)。

**Tech Stack:** TypeScript · Bun ≥ 1.3.13 · `bun test` · **无新第三方依赖**(纯 `node:` + 现有类型)。

## Global Constraints

> 每个任务的要求都隐含包含本节。数值/口径逐字照 spec/底稿。

- **CC-Haha 可直接复制/抄/移植/改写**:`<system-reminder>` 包壳、steering 队列语义按 cc-haha 源码/边界测试实现;许可允许复制/修改/发布。**行为对齐**:确定性逻辑(reminder 放置、steering drain、todo 归一)照 Python/cc-haha「同输入→同决策」。
- **todo 口径照我们 Python(非 cc-haha 字段)**:`TodoItem = { task: string; status: 'pending'|'in_progress'|'done' }`(status 用 `done`,**不是** cc-haha 的 `completed`;**无** `activeForm`)。渲染标记 `☐/◐/☑`。`todo_write` 工具 `read_only=false`、**不要审批**(它是本机可逆的进度记录)。
- **system-reminder 放置纪律(红线,别破)**:注入的提醒/steering/plan 说明是 `{role:'user', content: wrapReminder(...)}`,**只 append 到 messages 尾部,绝不 mutate 早前历史**;**只在一批 tool_call 的每个 role:tool 都已 append(配对完成)之后、或收尾时**注入——中途插一条 user 消息会破坏 OpenAI 兼容的 `tool_call_id` 配对 → provider 400。

- **循环从 `for` 改 `while`(可变 `turnsLimit`)**:每批 steering 给 `turnsLimit += STEER_EXTRA_TURNS(2)`,封顶 `maxTurns + Math.floor(maxTurns/2)`(既给新指令落地空间、又防"聊不完")。

```ts
// ts/src/types/todo.ts
export type TodoStatus = 'pending' | 'in_progress' | 'done'
export interface TodoItem {
  task: string
  status: TodoStatus
}

const TODO_MARK: Record<TodoStatus, string> = { pending: '☐', in_progress: '◐', done: '☑' }
const VALID_STATUS: readonly TodoStatus[] = ['pending', 'in_progress', 'done']

/** 渲染成大白话清单(照 Python format_todo_checklist)。 */
export function formatTodoChecklist(todos: TodoItem[]): string {
  const done = todos.filter(t => t.status === 'done').length
  const lines = todos.map(t => `${TODO_MARK[t.status]} ${t.task}`).join('\n')
  return `任务清单(共 ${todos.length} 步,已完成 ${done} 步):\n${lines}`
}

/** 归一模型给的 todos:接受 ["a","b"](全 pending)或 [{task|content, status}]。非法项跳过,永不抛。 */
export function normalizeTodos(raw: unknown): TodoItem[] {
  if (!Array.isArray(raw)) return []
  const out: TodoItem[] = []
  for (const item of raw) {
    if (typeof item === 'string') {
      if (item.trim()) out.push({ task: item.trim(), status: 'pending' })
      continue
    }
    if (item && typeof item === 'object') {
      const o = item as Record<string, unknown>
      const task = typeof o.task === 'string' ? o.task : typeof o.content === 'string' ? o.content : ''
      if (!task.trim()) continue
      const status = VALID_STATUS.includes(o.status as TodoStatus) ? (o.status as TodoStatus) : 'pending'
      out.push({ task: task.trim(), status })
    }
  }
  return out
}

/** 解析 markdown 勾选清单(task_progress 内联参数用);markdown 只能表达勾没勾 → done/pending。永不抛。 */
export function parseProgressMarkdown(md: string): TodoItem[] {
  if (typeof md !== 'string') return []
  const out: TodoItem[] = []
  for (const line of md.split('\n')) {
    const m = line.match(/^\s*[-*]\s*\[([ xX])\]\s*(.+?)\s*$/)
    if (m) out.push({ task: m[2]!, status: m[1]!.toLowerCase() === 'x' ? 'done' : 'pending' })
  }
  return out
}

// ts/src/harness/reminders.ts
import type { Message } from '../types/message'
import type { ToolContext } from '../tools/Tool'

export const STEER_MARK = '[用户补充/纠偏]'
export const STEER_EXTRA_TURNS = 2
export const PROGRESS_REMIND_EVERY = 6
export const PLAN_MODE_REMINDER =
  '你现在处于【计划模式】:只规划、不动手。用只读工具去探索,把完整、分步的计划讲清楚给老板;会实际改动的步骤先别做,等老板切到执行档或确认后再做。'

/** cc-haha 式系统提醒包壳:系统提示已告诉模型 <system-reminder> 是系统自动加的、不是老板说的话。 */
export function wrapReminder(content: string): string {
  return `<system-reminder>\n${content}\n</system-reminder>`
}

/**
 * 取空 steerInbox 的积压插话,作 [用户补充/纠偏] 用户消息 append 到 messages 尾部,返回取出的原文。
 * 原地 splice(路由持同一数组引用);空则不动 messages。只在安全点(批配对完/收尾)调用。
 */
export function drainSteering(messages: Message[], ctx: ToolContext): string[] {
  const inbox = ctx.steerInbox
  if (!inbox || inbox.length === 0) return []
  const drained = inbox.splice(0)
  for (const m of drained) messages.push({ role: 'user', content: `${STEER_MARK} ${m}` })
  return drained
}

/** steering 续命:每批插话给 turnsLimit 加 STEER_EXTRA_TURNS,封顶 maxTurns + floor(maxTurns/2)。 */
export function extendTurns(turnsLimit: number, maxTurns: number, batches: number): number {
  const cap = maxTurns + Math.floor(maxTurns / 2)
  return Math.min(cap, turnsLimit + STEER_EXTRA_TURNS * batches)
}

/** 本轮该注入哪些系统提醒(进度提醒 + plan 说明)。纯读——进度计数清零由循环负责(见 Task 4)。 */
export function collectReminders(ctx: ToolContext): Array<{ kind: 'progress' | 'plan'; text: string }> {
  const out: Array<{ kind: 'progress' | 'plan'; text: string }> = []
  if ((ctx.requestsSinceProgress ?? 0) >= PROGRESS_REMIND_EVERY) {
    const todos = ctx.todos ?? []
    const pct = todos.length ? Math.round((todos.filter(t => t.status === 'done').length / todos.length) * 100) : 0
    out.push({
      kind: 'progress',
      text: `你已经连着调了 ${PROGRESS_REMIND_EVERY} 次工具没更新任务清单。若在做多步任务,记得用 todo_write 更新进度(当前约 ${pct}%),别让老板和你自己跟丢了。`,
    })
  }
  if (ctx.permissionMode === 'plan') {
    out.push({ kind: 'plan', text: PLAN_MODE_REMINDER })
  }
  return out
}

export async function* runAgentLoop(opts: RunAgentLoopOptions): AsyncGenerator<AgentEvent> {
  const { model, registry } = opts
  const maxTurns = opts.maxTurns ?? 12
  const ctx: ToolContext = {
    workspace: opts.workspace,
    signal: opts.signal,
    sandbox: opts.sandbox,
    permissionMode: opts.permissionMode ?? 'ask',
    conversationId: opts.conversationId,
    autoSpendCount: 0,
    steerInbox: opts.steerInbox ?? [],
    todos: [],
    requestsSinceProgress: 0,
  }
  const messages: Message[] = [
    { role: 'system', content: opts.systemPrompt },
    { role: 'user', content: opts.userMessage },
  ]

  let turnsLimit = maxTurns
  let turn = 0
  while (turn < turnsLimit) {
    const step = await model.step({ messages, tools: registry.specs() })
    if (step.kind === 'final') {
      messages.push({ role: 'assistant', content: step.text })
      // steering 优先于收尾:模型想结束但收件箱有插话 → 灌进去、别收尾,接着跑
      const drained = drainSteering(messages, ctx)
      if (drained.length) {
        for (const m of drained) yield { type: 'steering', content: m }
        turnsLimit = extendTurns(turnsLimit, maxTurns, drained.length)
        turn++
        continue
      }
      yield { type: 'final', text: step.text }
      return
    }
    if (step.text) yield { type: 'thinking', text: step.text }
    messages.push({ role: 'assistant', content: step.text ?? '', toolCalls: step.calls })
    for (const call of step.calls) {
      yield { type: 'tool_call', tool: call.name, input: call.input }
      yield* gateOneCall(registry, call, ctx, messages)
    }
    // 一批工具全配对后(安全点):drain 插话灌进去,给下一次 model.step 看
    const drained = drainSteering(messages, ctx)
    if (drained.length) {
      for (const m of drained) yield { type: 'steering', content: m }
      turnsLimit = extendTurns(turnsLimit, maxTurns, drained.length)
    }
    turn++
  }

  // max_turns 兜底:强制一次无工具收敛(照 loop.py 的 _FINAL_NUDGE 哲学)。
  const forced = await model.step({ messages, tools: [] })
  yield { type: 'final', text: forced.kind === 'final' ? forced.text : '(已达最大轮次,未能收敛)' }
}
```
