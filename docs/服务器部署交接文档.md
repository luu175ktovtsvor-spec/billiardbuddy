### 步骤 2：配置 Nginx 反向代理

创建 Nginx 配置文件：

```bash
cat > /etc/nginx/sites-available/billiards << 'EOF'
server {
    listen 80 default_server;
    server_name _;

    # 前端
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }

    # 后端 API
    location /api/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 300s;
        add_header X-Accel-Buffering no;
    }

    # SSE 流式端点
    location /api/v1/stream/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Connection '';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 600s;
        add_header X-Accel-Buffering no;
    }

    # 上传文件静态服务
    location /uploads/ {
        alias /var/www/billiards-ai/server/uploads/;
        expires 7d;
        add_header Cache-Control "public, immutable";
    }
}
EOF

# 启用配置
ln -sf /etc/nginx/sites-available/billiards /etc/nginx/sites-enabled/billiards
rm -f /etc/nginx/sites-enabled/default
nginx -t && systemctl restart nginx
```

### 步骤 3：配置 systemd 后端服务

```bash
cat > /etc/systemd/system/billiards-backend.service << 'EOF'
[Unit]
Description=Billiards AI Backend (FastAPI)
After=network.target postgresql.service

[Service]
Type=simple
User=root
WorkingDirectory=/var/www/billiards-ai/server
Environment=PATH=/root/.local/bin:/usr/local/bin:/usr/bin:/bin
ExecStart=/root/.local/bin/uv run uvicorn main:app --host 0.0.0.0 --port 8000 --workers 2
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF
```

### 步骤 4：配置 systemd 前端服务

```bash
cat > /etc/systemd/system/billiards-frontend.service << 'EOF'
[Unit]
Description=Billiards AI Frontend (Next.js)
After=network.target billiards-backend.service

[Service]
Type=simple
User=root
WorkingDirectory=/var/www/billiards-ai/web
Environment=PORT=3000
Environment=HOSTNAME=0.0.0.0
Environment=NEXT_PUBLIC_API_URL=http://39.106.214.21:8000
ExecStart=/usr/bin/node .next/standalone/server.js
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF
```

### 步骤 8：开放阿里云安全组

在阿里云控制台 → ECS → 安全组，添加入方向规则：

| 协议 | 端口 | 授权对象 | 说明 |
|------|------|---------|------|
| TCP | 80 | 0.0.0.0/0 | HTTP 访问 |
| TCP | 443 | 0.0.0.0/0 | HTTPS（后续域名+SSL用） |

> 注意：不要开放 3000、8000、5432 端口给公网，通过 Nginx 80 端口转发即可。
